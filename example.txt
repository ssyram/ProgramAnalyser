real x : [0, 12] = 0;
int y : [-100, 5] = 0;
real z : [-100, 5] = 0;
random r1 : [0,1] ~ uniform(0, 1);
random r2 : [0.5,1] ~ uniform(0.5, 1);
[ true ]
while x < 10 do
    score(x);
    if y < 5 then
        if prob(0.8) then
            y := y + 1
        else
            y := y - 1;
            break
        fi
    else skip fi;
    if z < 5 then
        if prob(0.5) then
            z := z + r1
        else
            z := z - r1
        fi
    else skip
    fi;
    x := x + r2
od;
if y >= 5 then
    score(x)
else score(y)
fi
